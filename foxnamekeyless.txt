-- File cấu hình lưu key
local SAVE_FILE_NAME = "FoxName_KeyCache.json"
local CORRECT_KEY = "FoxNameKeyGPSHsow - Hsjskwoeoirieidikepapl"
local LINK_GET_KEY = "https://link4m.net/VlM6B"

-- Hàm kiểm tra thời gian lưu key (24 giờ = 86400 giây)
local function checkSavedKey()
    if readfile and writefile and isfile and isfile(SAVE_FILE_NAME) then
        local success, data = pcall(function()
            return game:GetService("HttpService"):JSONDecode(readfile(SAVE_FILE_NAME))
        end)
        if success and data and data.key == CORRECT_KEY then
            local currentTime = os.time()
            -- Kiểm tra xem đã qua 24 giờ (86400 giây) chưa
            if (currentTime - data.time) < 86400 then
                return true -- Key vẫn còn hạn
            end
        end
    end
    return false
end

-- Hàm lưu key kèm thời gian hiện tại
local function saveKey()
    if writefile then
        local data = {
            key = CORRECT_KEY,
            time = os.time()
        }
        pcall(function()
            writefile(SAVE_FILE_NAME, game:GetService("HttpService"):JSONEncode(data))
        end)
    end
end

-- Nếu key đã được lưu và còn hạn trong 24h thì chạy thẳng script gốc, không bật menu
if checkSavedKey() then
    local success, err = pcall(function()
        loadstring(game:HttpGet("https://raw.githubusercontent.com/caomod2077/Script/refs/heads/main/Fn-stealanegg.lua"))()
    end)
    if not success then
        warn("Error running script: " .. tostring(err))
    end
    return -- Dừng code, không tạo GUI nữa
end

-- Hàm tạo hiệu ứng Lắc (Shake Animation) mượt mà khi sai key
local function shakeMenu()
    local originalPos = UDim2.new(0.5, -240, 0.5, -135)
    local intensity = 10
    
    for i = 1, 4 do
        TweenService:Create(MainFrame, TweenInfo.new(0.04), {Position = originalPos + UDim2.new(0, intensity, 0, 0)}):Play()
        task.wait(0.04)
        TweenService:Create(MainFrame, TweenInfo.new(0.04), {Position = originalPos + UDim2.new(0, -intensity, 0, 0)}):Play()
        task.wait(0.04)
        intensity = intensity * 0.6
    end
    TweenService:Create(MainFrame, TweenInfo.new(0.05), {Position = originalPos}):Play()
end
